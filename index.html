async function trySupabaseLogin(identifier,password){ if(!supabaseClient){ return { ok:false, error:"Supabase não foi inicializado no navegador." }; } let email=String(identifier||"").trim(); if(!email.includes("@")){ const localUser=users.find(u=> String(u.username||"").toLowerCase()=== email.toLowerCase() ); if(localUser?.email){ email=String(localUser.email).trim(); }else{ return { ok:false, notFound:true, error:"Nome de utilizador não encontrado localmente." }; } } try{ const {data,error}=await supabaseClient.auth.signInWithPassword({ email, password }); if(error){ console.error( "ERRO REAL DO SUPABASE LOGIN:", error ); return { ok:false, error:error.message||"Erro de autenticação no Supabase." }; } if(!data?.session||!data?.user){ return { ok:false, error:"O Supabase não devolveu uma sessão válida." }; } cloudAuthenticated=true; setOnlineStatus("online"); const authUser=data.user; let profile=users.find(u=> u.email&& String(u.email).toLowerCase()=== String(authUser.email||"").toLowerCase() ); if(profile){ currentUser=profile; }else{ currentUser={ id:authUser.id, name:authUser.user_metadata?.name||authUser.email, username:authUser.user_metadata?.username||authUser.email, email:authUser.email, role:authUser.user_metadata?.role||"inspector", active:true, createdAt:new Date().toISOString() }; users.push(currentUser); saveUsers(); } localStorage.setItem( SESSION_KEY, currentUser.id ); return { ok:true, user:authUser }; }catch(error){ console.error( "ERRO EXCEPCIONAL NO LOGIN SUPABASE:", error ); return { ok:false, error:error?.message||String(error) }; } } async function login(username,password){ const normalized=String(username||"").trim(); const pass=password||""; if(!normalized||!pass){ return { ok:false, error:"Preencha o utilizador e a palavra-passe." }; } /* ========================================================= SE FOR E-MAIL: LOGIN DIRETO NO SUPABASE Não fazemos fallback silencioso para o login local. Assim conseguimos ver o erro verdadeiro. ========================================================= */ if(normalized.includes("@")){ const result= await trySupabaseLogin( normalized, pass ); if(result.ok){ return { ok:true }; } /* Se o Supabase recusou o e-mail, mostramos o erro real em vez de esconder. */ return { ok:false, error: result.error|| "Não foi possível autenticar no Supabase." }; } /* ========================================================= SE FOR NOME DE UTILIZADOR: procuramos o e-mail associado localmente e tentamos autenticar no Supabase. ========================================================= */ const localUser=users.find(u=> String(u.username||"").toLowerCase()=== normalized.toLowerCase() ); if(localUser?.email){ const result= await trySupabaseLogin( localUser.email, pass ); if(result.ok){ return { ok:true }; } /* Se existe um e-mail associado mas o Supabase recusou, não escondemos o erro. */ return { ok:false, error: result.error|| "Não foi possível autenticar o utilizador." }; } /* ========================================================= LOGIN ADMINISTRATIVO LOCAL ========================================================= */ const localLoginUser=users.find(u=> ( String(u.username||"").toLowerCase()=== normalized.toLowerCase() )&& String(u.password||"")===pass&& u.active!==false ); if(localLoginUser){ currentUser=localLoginUser; cloudAuthenticated=false; setOnlineStatus("offline"); localStorage.setItem( SESSION_KEY, localLoginUser.id ); return { ok:true }; } if( normalized.toLowerCase()==="admin"&& pass==="admin123" ){ let admin=users.find( u=>u.role==="admin" ); if(admin){ admin.username="admin"; admin.password="admin123"; admin.active=true; if(!admin.name){ admin.name="Administrador"; } if(!admin.email){ admin.email="admin@foodsafemoz.local"; } saveUsers(); currentUser=admin; }else{ admin={ id:"u_admin", name:"Administrador", username:"admin", email:"admin@foodsafemoz.local", password:"admin123", role:"admin", active:true, createdAt:new Date().toISOString() }; users.push(admin); saveUsers(); currentUser=admin; } cloudAuthenticated=false; setOnlineStatus("offline"); localStorage.setItem( SESSION_KEY, currentUser.id ); return { ok:true }; } return { ok:false, error:"Utilizador local não encontrado." }; } async function doLogin(){ const username= document.getElementById("loginUser").value.trim(); const password= document.getElementById("loginPass").value; const error= document.getElementById("loginError"); if(!username||!password){ error.textContent= "Preencha o utilizador e a palavra-passe."; error.classList.remove("hidden"); return; } error.classList.add("hidden"); const result= await login(username,password); if(!result.ok){ console.error( "FoodSafe Login:", result.error ); /* Mostra temporariamente o erro real. Isso vai permitir identificar exatamente o que o Supabase está a devolver. */ error.textContent= "Erro de login: "+ (result.error||"Falha desconhecida."); error.classList.remove("hidden"); return; } openApp(); if(cloudAuthenticated){ await syncAllFromCloud(); } }
